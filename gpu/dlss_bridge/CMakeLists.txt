cmake_minimum_required(VERSION 3.24)
project(order1886_dlss LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build this optional DLL separately with x64 MSVC; the host may use MinGW")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ORDER_DLSS_SDK "" CACHE PATH "Official NVIDIA DLSS SDK 310.9.16 root (outside tracked sources)")
find_path(ORDER_VULKAN_INCLUDE vulkan/vulkan.h REQUIRED)
include(CTest)
add_executable(order_dlss_contract_tests tests/contracts.cpp)
target_include_directories(order_dlss_contract_tests PRIVATE "${ORDER_VULKAN_INCLUDE}" .)
target_compile_definitions(order_dlss_contract_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME order_dlss_contracts COMMAND order_dlss_contract_tests)
if(NOT EXISTS "${ORDER_DLSS_SDK}/include/nvsdk_ngx_vk.h")
    message(FATAL_ERROR "ORDER_DLSS_SDK must name the separately downloaded official SDK 310.9.16")
endif()
add_library(order1886_dlss SHARED order1886_dlss.cpp)
target_include_directories(order1886_dlss PRIVATE "${ORDER_VULKAN_INCLUDE}" "${ORDER_DLSS_SDK}/include")
target_compile_definitions(order1886_dlss PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(order1886_dlss PRIVATE
    "${ORDER_DLSS_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib" Version Shlwapi Advapi32)
target_compile_options(order1886_dlss PRIVATE /W4 /EHsc)
add_executable(order_dlss_abi_tests tests/abi.cpp)
target_include_directories(order_dlss_abi_tests PRIVATE "${ORDER_VULKAN_INCLUDE}")
target_compile_definitions(order_dlss_abi_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME order_dlss_abi COMMAND order_dlss_abi_tests "$<TARGET_FILE:order1886_dlss>")
add_executable(order_dlss_lifecycle_tests tests/lifecycle.cpp order1886_dlss.cpp)
target_include_directories(order_dlss_lifecycle_tests PRIVATE "${ORDER_VULKAN_INCLUDE}" "${ORDER_DLSS_SDK}/include")
target_compile_definitions(order_dlss_lifecycle_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
# The test supplies NGX and Vulkan entry points; it never links the NVIDIA runtime/library.
add_test(NAME order_dlss_lifecycle COMMAND order_dlss_lifecycle_tests)
add_executable(order_dlss_gpu_smoke EXCLUDE_FROM_ALL tests/gpu_smoke.cpp)
add_dependencies(order_dlss_gpu_smoke order1886_dlss)
target_include_directories(order_dlss_gpu_smoke PRIVATE "${ORDER_VULKAN_INCLUDE}" "../shim")
target_compile_definitions(order_dlss_gpu_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN VK_NO_PROTOTYPES)
target_link_libraries(order_dlss_gpu_smoke PRIVATE Bcrypt Version)
